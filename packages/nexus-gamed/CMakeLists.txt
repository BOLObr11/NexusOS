cmake_minimum_required(VERSION 3.20)
project(nexus_gamed VERSION 0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_compile_options(-Wall -Wextra -Wpedantic -Werror -O2)
add_executable(nexus-gamed src/daemon.cpp)
add_executable(nexusctl src/client.cpp)
install(TARGETS nexus-gamed RUNTIME DESTINATION libexec/nexus)
install(TARGETS nexusctl RUNTIME DESTINATION bin)
